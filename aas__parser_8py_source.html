14def get_submodel_ids(shell: model.AssetAdministrationShell) -> list[str]:
15 """Get all IDs from the submodels referenced in the given AAS.
16
17 :param shell: The Asset Administration Shell to extract submodel IDs from.
18 :return: A list of submodel IDs referenced in the AAS.
19 """
20 submodel_ids = []
21 for submodel in shell.submodel:
22 if len(submodel.key) < 1 or submodel.key[0].type != model.KeyTypes.SUBMODEL:
23 _logger.warning(f"Submodel reference {submodel} does not start with SUBMODEL key type.")
24 continue
25
26 submodel_ids.append(submodel.key[0].value)
27
28 return submodel_ids

31def parse_environment_file(file_path: str) -> EnvironmentData | None:
32 """Parse an environment file and return an EnvironmentData instance.
33
34 :param file_path: The path to the environment file to parse.
35 :return: An EnvironmentData instance containing the parsed data.
36 """
37 file = Path(file_path)
38 if not file.exists():
39 raise FileNotFoundError(f"Submodel structure file not found: {file}")
40
41 try:
42 with file.open("r", encoding="utf-8") as f:
43 json_content = json.load(f)
44 except json.JSONDecodeError as e:
45 _logger.error(f"Failed to parse JSON file '{file}': {e}")
46 return None
47
48 if not json_content or not isinstance(json_content, dict):
49 _logger.error(f"File '{file}' does not contain valid environment data.")
50 return None
51
52 env_content = EnvironmentData(file_name=file.name, file=file)
53 env_content.submodels = _parse_node("submodels", json_content)
54 env_content.concept_descriptions = _parse_node("conceptDescriptions", json_content)
55 env_content.shells = _parse_node("assetAdministrationShells", json_content)
56
57 return env_content

60def get_description_from_shell(shell: model.AssetAdministrationShell, language: str = "en") -> str | None:
61 """Get the description from an Asset Administration Shell.
62
63 :param shell: The Asset Administration Shell to extract the description from.
64 :param language: The language code for the description.
65 :return: The description string if found, otherwise None.
66 """
67 if shell.description is None:
68 _logger.warning(f"No description found for shell {shell.id_short}")
69 return None
70
71 keys = shell.description.keys()
72
73 if keys is None or len(keys) == 0:
74 _logger.warning(f"No description keys found for shell {shell.id_short}")
75 return None
76
77 if language not in keys:
78 _logger.warning(f"Description for language '{language}' not found in shell {shell.id_short}")
79 return None
80
81 return shell.description.get(language)
