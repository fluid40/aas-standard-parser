12def get_submodel_ids(shell: model.AssetAdministrationShell) -> list[str]:
13 """Get all IDs from the submodels referenced in the given AAS.
14
15 :param shell: The Asset Administration Shell to extract submodel IDs from.
16 :return: A list of submodel IDs referenced in the AAS.
17 """
18 submodel_ids = []
19 for submodel in shell.submodel:
20 if len(submodel.key) < 1 or submodel.key[0].type != model.KeyTypes.SUBMODEL:
21 _logger.warning(f"Submodel reference {submodel} does not start with SUBMODEL key type.")
22 continue
23
24 submodel_ids.append(submodel.key[0].value)
25
26 return submodel_ids
