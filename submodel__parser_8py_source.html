58def get_submodel_element_by_id_short_path(submodel: model.Submodel, id_short_path: str) -> model.SubmodelElement:
59 """Retrieve a specific submodel element from the submodel at a specific idShort path.
60
61 :param submodel: The submodel to search within.
62 :param id_short_path: IdShort path to the submodel element (dot-separated), e.g., "Element1.Element2[0].Element3".
63 :return: The found submodel element or None if not found.
64 """
65 # Split the path by '.' and traverse the structure
66 parts = id_short_path.split(".")
67 current_elements = submodel.submodel_element
68 part_index = 0
69 for part in parts:
70 part_index += 1
71 # Handle indexed access like "Element[0]" for SubmodelElementLists
72 if "[" in part and "]" in part:
73 # Split SubmodelElementList name and index
74 base, idx = part[:-1].split("[")
75 idx = int(idx)
76 # Find the SubmodelElementList in the current elements
77 submodel_element = next((el for el in current_elements if el.id_short == base), None)
78
79 if not submodel_element or not (isinstance(submodel_element, (model.SubmodelElementList, model.SubmodelElementCollection))):
80 logger.debug(f"Submodel element '{base}' not found or is not a collection/list in current {current_elements}.")
81 return None
82
83 # Check if index is within range
84 if idx >= len(submodel_element.value):
85 logger.debug(f"Index '{idx}' out of range for element '{base}' with length {len(submodel_element.value)}.")
86 return None
87
88 # get the element by its index from SubmodelElementList
89 submodel_element = submodel_element.value[idx]
90
91 else:
92 # Find the SubmodelElement in the current SubmodelElementCollection
93 submodel_element = next((el for el in current_elements if el.id_short == part), None)
94
95 if not submodel_element:
96 logger.debug(f"Submodel element '{part}' not found in current {current_elements}.")
97 return None
98
99 # If we've reached the last part, return the found element
100 if part_index == len(parts):
101 return submodel_element
102
103 # If the found element is a collection or list, continue traversing
104 if isinstance(submodel_element, (model.SubmodelElementCollection, model.SubmodelElementList)):
105 current_elements = submodel_element.value
106 else:
107 return submodel_element
108
109 return submodel_element
