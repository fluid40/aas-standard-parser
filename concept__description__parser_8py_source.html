13def parse_concept_description(
14 concept_description: model.concept.ConceptDescription, preferred_languages: list[str] | None = None
15) -> ConceptDescriptionData:
16 """Parse a Concept Description into a ConceptDescriptionData containing its display name and description.
17
18 :param concept_description: The Concept Description to parse.
19 :param preferred_languages: A list of preferred languages to extract the display name and description from, in order of preference. Defaults to ["en", "de"].
20 :return: A ConceptDescriptionData with attributes "display_name" and "description" if found.
21 """
22 display_name = get_display_name_from_concept_description(concept_description, preferred_languages=preferred_languages)
23 description = get_description_from_concept_description(concept_description, preferred_languages=preferred_languages)
24
25 concept_description_data = ConceptDescriptionData(concept_description.id, concept_description.id_short)
26 concept_description_data.main_display_name = display_name if display_name is not None else ""
27 concept_description_data.main_description = description if description is not None else ""
28 concept_description_data.display_name = multi_language_parser.parse_multi_language_object_to_dict(concept_description.display_name)
29 concept_description_data.description = multi_language_parser.parse_multi_language_object_to_dict(concept_description.description)
30
31 data_specification = concept_description.embedded_data_specifications
32
33 for embedded_data_spec in data_specification:
34 data_spec_content = DataSpecificationContent()
35 content = embedded_data_spec.data_specification_content
36
37 if not isinstance(content, model.DataSpecificationIEC61360):
38 continue
39
40 ice_content: model.DataSpecificationIEC61360 = content
41
42 data_spec_content.unit = ice_content.unit if ice_content.unit is not None else ""
43 data_spec_content.unit_id = ice_content.unit_id.key[0].value if ice_content.unit_id is not None else ""
44 data_spec_content.value = content.value if content.value is not None else ""
45 data_spec_content.value_format = content.value_format if content.value_format is not None else ""
46 data_spec_content.value_list = list(content.value_list) if content.value_list is not None else []
47 data_spec_content.symbol = content.symbol if content.symbol is not None else ""
48 data_spec_content.definition = multi_language_parser.parse_multi_language_object(content.definition) if content.definition is not None else ""
49
50 data_spec_content.data_type = {"name": content.data_type.name} if content.data_type is not None else {}
51 data_spec_content.short_name = multi_language_parser.parse_multi_language_object(content.short_name) if content.short_name is not None else ""
52 data_spec_content.preferred_name = (
53 multi_language_parser.parse_multi_language_object(content.preferred_name) if content.preferred_name is not None else ""
54 )
55 # Populate the data_spec_content with relevant information from content if needed
56 concept_description_data.data_specification_content.append(data_spec_content)
57
58 return concept_description_data

62 """Parse a Concept Description from a dictionary into a ConceptDescriptionData containing its display name and description.

64 :param concept_description_dict: The dictionary representing the Concept Description to parse.
