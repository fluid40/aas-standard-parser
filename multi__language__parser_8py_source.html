7 """Extract a readable text from BaSyx multilingual text/name containers.
8
9 :param multi_language_object: A dictionary containing multilingual text with language codes as keys.
10 :param preferred_languages: A list of preferred languages to extract the text from, in order of preference. Defaults to ["en", "de"].
11 :return: The extracted text in the preferred language, or the first available text if none of the preferred languages are found, or an empty string if no text is available.
12 """
13 if preferred_languages is None:
14 preferred_languages = ["en", "de"]
15
16 if multi_language_object is None:
17 return ""
